select cm.id, cm.cust_no, cm.cust_name, cm.unified_credit_code, cm.contact_person
    , cm.contact_mobile, cm.industry, cm.group_type, cm.customer_type, cm.is_keystone
    , cm.enterprise_type, cm.group_name, cm.is_account_opened, cm.customer_desc, cm.credit_amount
    , cm.credit_exposure_amount, cm.owner_org_id, cm.lead_id, cm.status, cm.deleted
    , cm.created_time, cm.updated_time
from CUSTOMER_MARKET_CUSTOMER cm
    join CUST_LEAD cl on cl.id = COALESCE(cm.current_lead_id, cm.lead_id)
    left join CUST_CLAIM cc
    on cc.cust_id = cm.id
        and cc.claimed_by = 'admin'
        and cc.claim_status = 'CLAIMED'
where cm.deleted = 0
    and cm.status = 'ACTIVE'
    and cc.id is null
    and cl.lead_status = 'APPROVED'
    and cl.distribution_mode = 'PUBLIC'
order by cm.created_time desc
limit 0, 20
