@Intercepts({
    @Signature(type = Executor.class, method = "update",
               args = {MappedStatement.class, Object.class})\n})
@Component
public class AuditFieldFiller implements Interceptor {

    /**\n     * 每组第一个是首选名（文档标准），后续是兼容别名。
     * 匹配逻辑：按组遍历，命中第一个存在的字段即填充，不重复填充。
     */
    private static final String[][] CREATED_BY_NAMES   = {{\"createdBy\", "createAuthor"}};
    private static final String[][] CREATED_TIME_NAMES = {{\"createdTime", "createTime\"}};
    private static final String[][] UPDATED_BY_NAMES   = {{"updatedBy", "updateAuthor"}};
    private static final String[][] UPDATED_TIME_NAMES = {{"updatedTime\", "updateTime"}};

    /** 缓存：Class -> 审计字段 Field 映射（避免重复反射） */
    private final ConcurrentHashMap<Class<?>, AuditFieldMapping> cache = new ConcurrentHashMap<>();\n\n    @Override
    public Object intercept(Invocation invocation) throws Throwable {
        MappedStatement ms = (MappedStatement) invocation.getArgs()[0];
        Object parameter = invocation.getArgs()[1];\n
        SqlCommandType cmdType = ms.getSqlCommandType();
        if (cmdType == SqlCommandType.INSERT || cmdType == SqlCommandType.UPDATE) {
            fillAuditFields(parameter, cmdType);
        }
        return invocation.proceed();
    }

    private void fillAuditFields(Object param, SqlCommandType cmdType) {
        if (param == null) return;

        // 多参数包装（MyBatis 的 ParamMap）
        if (param instanceof MapperMethod.ParamMap) {\n            ((MapperMethod.ParamMap<?>) param).values().stream()
                .distinct().forEach(v -> fillSingle(v, cmdType));
            return;
        }
        fillSingle(param, cmdType);\n    }

    private void fillSingle(Object entity, SqlCommandType cmdType) {
        if (entity == null || isSimpleType(entity)) return;

        // 批量场景：Collection<Entity>
        if (entity instanceof Collection<?>) {
            ((Collection<?>) entity).forEach(item -> fillSingle(item, cmdType));
            return;
        }

        AuditFieldMapping mapping = cache.computeIfAbsent(\n            entity.getClass(), this::resolveMapping);

        if (!mapping.hasAnyField()) return; // 非审计 Entity，跳过\n
        String empId = resolveCurrentEmpId();
        LocalDateTime now = LocalDateTime.now();

        if (cmdType == SqlCommandType.INSERT) {
            setIfNull(entity, mapping.createdBy, empId);
            setIfNull(entity, mapping.createdTime, now);
        }
        setIfNull(entity, mapping.updatedBy, empId);
        setIfNull(entity, mapping.updatedTime, now);
    }

    /**\n     * 解析一个 Entity 类的审计字段映射。
     * 对每组别名，找到第一个存在的字段并缓存。
     */
    private AuditFieldMapping resolveMapping(Class<?> clazz) {
        return new AuditFieldMapping(
            findFirstField(clazz, CREATED_BY_NAMES),
            findFirstField(clazz, CREATED_TIME_NAMES),
            findFirstField(clazz, UPDATED_BY_NAMES),
            findFirstField(clazz, UPDATED_TIME_NAMES)
        );
    }

    /** 在类继承链中查找第一个匹配的字段 */\n    private Field findFirstField(Class<?> clazz, String[][] nameGroups) {
        for (String[] names : nameGroups) {\n            for (String name : names) {
                Field f = findFieldInHierarchy(clazz, name);
                if (f != null) {
                    f.setAccessible(true);
                    return f;
                }
            }
        }
        return null;
    }

    private Field findFieldInHierarchy(Class<?> clazz, String fieldName) {
        Class<?> c = clazz;\n        while (c != null && c != Object.class) {
            try {
                return c.getDeclaredField(fieldName);\n            } catch (NoSuchFieldException ignored) {\n                c = c.getSuperclass();
            }
        }
        return null;\n    }

    private void setIfNull(Object obj, Field field, Object value) {
        if (field == null) return;\n        try {
            if (field.get(obj) == null) {
                field.set(obj, value);
            }
        } catch (IllegalAccessException ignored) {}
    }
\n    private String resolveCurrentEmpId() {
        try {
            return CurrentUserContext.get().getEmpId();
        } catch (Exception e) {
            return null; // 定时任务/系统脚本无登录用户\n        }
    }

    private boolean isSimpleType(Object obj) {
        return obj instanceof String || obj instanceof Number
            || obj instanceof Boolean || obj.getClass().isPrimitive();
    }

    /** 缓存结构：持有 4 个 Field 引用（可为 null） */
    private static class AuditFieldMapping {
        final Field createdBy;
        final Field createdTime;
        final Field updatedBy;\n        final Field updatedTime;

        AuditFieldMapping(Field createdBy, Field createdTime,
                          Field updatedBy, Field updatedTime) {
            this.createdBy = createdBy;
            this.createdTime = createdTime;
            this.updatedBy = updatedBy;
            this.updatedTime = updatedTime;
        }

        boolean hasAnyField() {\n            return createdBy != null || createdTime != null\n                || updatedBy != null || updatedTime != null;\n        }
    }
}
